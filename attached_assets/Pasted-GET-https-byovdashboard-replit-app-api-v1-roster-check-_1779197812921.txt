Content-Type: application/json

Server trims whitespace, de-dupes, and tolerates empty strings.

Rules
Permanent wins: if a tech is on both rosters, you'll only see rosterType: "Permanent".
Skip ordering a new-hire van when either:
rosterType === "Permanent", or
rosterType === "NewHire" and intent === "Permanent"
Order a training van when rosterType === "NewHire" and intent === "Training_Only" (or null).
rosterType === "None" → tech isn't in either roster; handle per existing onboarding process.
Drop-in Node helper for the Nexus agent
const BYOV_URL = "https://byovdashboard.replit.app";
async function checkRoster(enterpriseIds) {
  const res = await fetch(`${BYOV_URL}/api/v1/roster-check/bulk`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-API-Key": process.env.BYOV_API_KEY,
    },
    body: JSON.stringify({ enterpriseIds }),
  });
  if (!res.ok) throw new Error(`Roster check failed: ${res.status} ${await res.text()}`);
  return res.json();
}
function needsNewHireVan(result) {
  if (result.rosterType === "Permanent") return false;
  if (result.rosterType === "NewHire" && result.intent === "Permanent") return false;
  return true;
}
